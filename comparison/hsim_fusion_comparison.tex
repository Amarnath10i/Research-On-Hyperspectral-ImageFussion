\begin{abstract}
Consolidated comparison of recent hyperspectral--multispectral (HSI-MSI) image
fusion methods (supervised, unsupervised/blind, zero-shot, model-based). Only
IEEE Xplore-indexed venues (TGRS, TIP, TCSVT, CVPR) are included; arXiv-only
and MDPI entries are omitted. Metadata and SOTA for the 11 papers marked
\textbf{[PDF]} were read from source PDFs; SOTA from rasterized tables was
recovered via OCR and is approximate ($\approx$). A broader 23-paper survey
(dataset-usage matrix) is maintained separately in \texttt{review/DATASETS.md}.
\end{abstract}

\section*{Legend}
\begin{itemize}
  \item \textbf{Type}: Sup = supervised; Unsup = unsupervised/blind;
        Zero = zero-shot; MB = model-based / unrolled.
  \item \textbf{Scale}: $\times4/\times8/\times16/\times32$ spatial upscaling;
        \texttt{s=8}/\texttt{s=32} denote the same; \texttt{pansharp} = pansharpening.
  \item \textbf{Indicator}: \texttt{exact} = quoted from paper; $\approx$ \texttt{OCR}
        = approximate (rasterized table OCR'd); $\Delta$ = only improvement over a
        baseline reported; \texttt{QNR} = real data, no-reference metric;
        \texttt{--} = not evaluated / see paper.
  \item \textbf{[PDF]} = metadata/SOTA read from the source PDF.
\end{itemize}

\section{Corrections vs. the provided ``verified'' BibTeX}
The supplied BibTeX claimed several DOIs/authors as verified; the source PDFs
contradict the following (now fixed in \texttt{hsim\_fusion\_comparison.bib}):
\begin{itemize}
  \item \texttt{wang2026equivariant}: DOI 10.1109/TIP.2026.3537982 and
        datasets CAVE/Chikusei $\to$ corrected DOI
        \textbf{10.1109/TIP.2026.3657219}, datasets \textbf{CAVE/ICVL}.
  \item \texttt{dou2026bhsrnet}: was TGRS with placeholder DOI and author
        ``Dou, Yujie et al.'' $\to$ \textbf{TIP 2026}, DOI
        \textbf{10.1109/TIP.2026.3714832}, authors \textbf{Mai Xu, Yongxuan
        Dou, Xin Deng, Xin Zou, Zhenwei Shi}.
  \item \texttt{li2026bfmm}: placeholder DOI / ``Li, Rui and others'' $\to$
        DOI \textbf{10.1109/TGRS.2026.3699818}, authors \textbf{Yan Li,
        Chuangjie Fang, \dots, Yi-Peng Liu}.
  \item \texttt{liu2026semfnet}: was TIP with author ``Xing, Yuan et al.''
        $\to$ \textbf{TGRS 2026 (5501916)}, DOI
        \textbf{10.1109/TGRS.2026.3653545}, authors \textbf{Siyuan Liu,
        Zezheng Zhang, \dots, Shuaiqi Liu}.
  \item \texttt{chen2025cyformer}: DOI 10.1109/TCSVT.2024.3456789 $\to$
        \textbf{10.1109/TCSVT.2024.3461829}.
  \item \texttt{wang2026shotun}: datasets CAVE/Harvard/Chikusei $\to$
        datasets are \textbf{PaviaC/KSC/SanDiego/Houston}.
\end{itemize}

\section*{Dataset conventions}
\begin{itemize}
  \item \textbf{Simulated (Wald protocol)}: CAVE ($\times$8/$\times$16/$\times$32),
        Harvard ($\times$8), Chikusei ($\times$8), Pavia/ICVL.
  \item \textbf{Real / no-GT}: PaviaU, Chikusei, Houston2018, WV-3, PRISMA,
        LN01, Ziyuan-1 — evaluated with no-reference metrics (QNR, D$\lambda$, Ds).
  \item \textbf{Pansharpening / mosaiced-PAN} methods (DM-ZS, rho-PNN,
        Equivariant) are cross-task and use PRISMA/WV-3/PanCollection.
\end{itemize}
